  function showView(name){
    document.querySelectorAll(".view").forEach(v => v.classList.remove("active"));
    const el = document.getElementById("view-" + name);
    if(el) el.classList.add("active");
    document.querySelectorAll(".nav-btn").forEach(b => {
      b.classList.toggle("active", b.getAttribute("data-view") === name);
    });
    if(name === "trips") loadTrips();
    if(name === "profile") loadProfile();
  }
  document.querySelectorAll(".nav-btn").forEach(btn => {
    btn.addEventListener("click", () => showView(btn.getAttribute("data-view")));
  });

  function setAppShellVisible(visible){
    document.getElementById("topbar").style.display = visible ? "flex" : "none";
    document.getElementById("bottom-nav").style.display = visible ? "flex" : "none";
  }

  // ========== AUTH ==========
  document.getElementById("send-otp-btn").addEventListener("click", async () => {
    const email = document.getElementById("login-email").value.trim();
    const banner = document.getElementById("login-banner");
    banner.className = "banner"; banner.textContent = "";
    if(!email){ banner.className = "banner err"; banner.textContent = "Entrez une adresse email valide."; return; }

    btn.disabled = true; btn.textContent = "Envoi en cours…";

  });

  sb.auth.onAuthStateChange((event, session) => {
    document.getElementById("loading-screen").style.display = "none";
    if(session){
      setAppShellVisible(true);
      showView("book");
      const d = document.getElementById("f-date");
      if(d && !d.value) d.value = new Date().toISOString().slice(0,10);
    } else {
      setAppShellVisible(false);
      document.getElementById("login-email").value = "";
      document.getElementById("login-banner").className = "banner";

    }
  });

  document.getElementById("book-form").addEventListener("submit", async (e) => {
    e.preventDefault();
    const banner = document.getElementById("book-banner");
    banner.className = "banner"; banner.textContent = "";

  });

  // ========== MES COURSES ==========
  async function loadTrips(){
    const list = document.getElementById("trips-list");
    const empty = document.getElementById("trips-empty");
    list.innerHTML = "";

    const { data: userData } = await sb.auth.getUser();
    const user = userData?.user;
    if(!user) return;

    const { data, error } = await sb
      .from("reservations")
      .select("*")
      .eq("user_id", user.id)
      .order("date_heure", { ascending: false });

    if(error || !data || data.length === 0){
      empty.style.display = "block";
      return;
    }
    empty.style.display = "none";

    data.forEach(r => {
      const dt = new Date(r.date_heure);
      const whenStr = dt.toLocaleDateString("fr-BE", { day:"2-digit", month:"short" }) + " · " +
                      dt.toLocaleTimeString("fr-BE", { hour:"2-digit", minute:"2-digit" });
      const statut = r.statut || "en_attente";
      const card = document.createElement("div");
      card.className = "course-card";
      const cancelBtn = statut === "en_attente"
        ? `<button class="btn btn-ghost cancel-btn" data-id="${r.id}" style="margin-top:12px;font-size:.82rem;padding:10px 16px;">Annuler cette course</button>`
        : "";
      card.innerHTML = `
        <div class="top-row">
          <span class="when">${whenStr}</span>
          <span class="status-pill ${statut}">${STATUS_LABELS[statut] || statut}</span>
        </div>
        <div class="course-route">
          <div class="stop depart"><span class="mark"></span>${escapeHtml(r.depart)}</div>
          <div class="connector"></div>
          <div class="stop arrivee"><span class="mark"></span>${escapeHtml(r.arrivee)}</div>
        </div>
        ${cancelBtn}
      `;
      list.appendChild(card);
    });

    document.querySelectorAll(".cancel-btn").forEach(btn => {
      btn.addEventListener("click", () => cancelTrip(btn.getAttribute("data-id"), btn));
    });
  }

  async function cancelTrip(id, btn){
    if(!confirm("Annuler cette course ?")) return;
    btn.disabled = true; btn.textContent = "Annulation…";
    const { error } = await sb
      .from("reservations")
      .update({ statut: "annulee" })
      .eq("id", id);
    if(error){
      alert("Erreur lors de l'annulation : " + error.message);
      btn.disabled = false; btn.textContent = "Annuler cette course";
      return;
    }
    loadTrips();
  }

  function escapeHtml(str){
    const div = document.createElement("div");
    div.textContent = str || "";
    return div.innerHTML;
  }

  // ========== PROFIL ==========
  async function loadProfile(){
    const { data: userData } = await sb.auth.getUser();
    const user = userData?.user;
    if(!user) return;

    document.getElementById("profile-email-display").textContent = user.email || "";
    document.getElementById("profile-initial").textContent = (user.email || "?").charAt(0).toUpperCase();

    const { data: profil } = await sb.from("profils").select("*").eq("id", user.id).maybeSingle();
    if(profil){
      document.getElementById("p-nom").value = profil.nom || "";
      document.getElementById("p-telephone").value = profil.telephone || "";
      document.getElementById("profile-name-display").textContent = profil.nom || "Sans nom";
    } else {
      document.getElementById("profile-name-display").textContent = "Sans nom";
    }
  }
